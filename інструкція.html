Інструкція розрахована на людину без технічного досвіду. Писати нічого не доведеться: увесь потрібний код — просто тут, на сторінці, з кнопкою «Скопіювати».

Познач крок галочкою, коли зробиш — сторінка запам'ятає, де ти зупинилась.

З чого складається помічник

Браузер
Руки. Ходить по work.ua під твоїм обліковим записом і збирає кандидатів.
Gemini
Голова. Читає резюме і вирішує, наскільки людина підходить.
Таблиця
Пам'ять. Зберігає завдання, результати і тих, кого вже показували.

Чому саме браузер, а не сама таблиця: work.ua не пускає до себе сервери Google — відповідає «перевірка надійності підключення». А твій браузер пускає, бо це ти.

02Ключ Gemini

5 хв

Gemini — штучний інтелект від Google, який читатиме резюме. Ключ — довгий пароль, за яким Google впізнає тебе. Безкоштовний, картка не потрібна.

  1. Відкрий aistudio.google.com/app/apikey і увійди.
  2. Натисни Create API key. Якщо спитає про проєкт — обери будь-який.
  3. З'явиться довгий рядок. Copy — і поклади в нотатки, щоб не загубити.

Ключ — це пароль. Не надсилай його в чати і не показуй на екрані. Якщо показала — видали на тій самій сторінці й створи новий, це хвилина.

Ключ зберігається тільки у твоєму браузері. Він не потрапляє ні в таблицю, ні на work.ua, ні до ведучої.

/**
 * Агент-рекрутер для work.ua — частина, що живе в Google Таблиці.
 *
 * Вона НЕ ходить на work.ua (Cloudflare не пускає сервери Google — див. README).
 * Її робота — віддавати завдання з аркуша «Пошуки», казати, кого вже бачили,
 * і записувати результат у «Кандидати», «Історію» та «Журнал».
 *
 * Ходить на work.ua і звертається до Gemini той, хто працює в браузері —
 * скрипт «агент.user.js» у Tampermonkey.
 *
 * ── Як увімкнути ─────────────────────────────────────────────────────
 * 1. Заміни значення КЛЮЧ нижче на свій вигаданий пароль.
 * 2. Розгорнути → Новий розгорток → тип «Веб-застосунок»
 *    Виконувати як: Я
 *    Хто має доступ: Усі

function doPost(e) {
  try {
    var запит = JSON.parse(e.postData.contents);

    if (запит.ключ !== КЛЮЧ) {
      return відповідь({ ok: false, помилка: 'Невірний ключ' });
    }

    switch (запит.дія) {
      case 'пошуки':
        return відповідь({ ok: true, пошуки: дайПошуки() });

      case 'історія':
        return відповідь({ ok: true, номери: дайІсторію(запит.пошук) });

      case 'зберегти':
        var підсумок = зберегти(запит.пошук, запит.кандидати || []);
        return відповідь({ ok: true, записано: підсумок.записано, аркуш: підсумок.аркуш });

      case 'очистити':
        return відповідь({ ok: true, видалено: очиститиІсторію(запит.пошук) });

      case 'журнал':
        записатиЖурнал(запит.рядок || {});
        return відповідь({ ok: true });

      case 'статус':
        оновитиСтатус(запит.пошук, запит.статус);
        return відповідь({ ok: true });

      default:
        return відповідь({ ok: false, помилка: 'Невідома дія: ' + запит.дія });
    }
  } catch (err) {
    return відповідь({ ok: false, помилка: String(err) });
  }
}

function аркуш(назва) {
  var sh = SpreadsheetApp.getActive().getSheetByName(назва);
  if (!sh) throw new Error('Не знайдено аркуш «' + назва + '»');
  return sh;
}

/**
 * Шукає рядок шапки (у перших 10 рядках) і повертає:
 *   { рядок: N, колонки: { 'Назва пошуку': 0, ... } }
 * Завдяки цьому колонки можна переставляти місцями — код не зламається.
 */
function шапкаЗа(sh, якір, підпис) {
  var скільки = Math.min(10, sh.getLastRow());
  if (скільки === 0) throw new Error('Аркуш «' + підпис + '» порожній');

  var зона = sh.getRange(1, 1, скільки, sh.getLastColumn()).getValues();

  for (var р = 0; р < зона.length; р++) {
    var клітинки = зона[р].map(function (з) { return String(з).trim(); });
    if (клітинки.indexOf(якір) !== -1) {
      var карта = {};
      клітинки.forEach(function (з, i) { if (з) карта[з] = i; });
      return { sh: sh, рядок: р + 1, колонки: карта };
    }
  }
  throw new Error('В аркуші «' + підпис + '» не знайдено колонку «' + якір + '»');
}

function шапка(назваАркуша) {
  return шапкаЗа(аркуш(назваАркуша), ЯКОРІ[назваАркуша], назваАркуша);
}

var СЛУЖБОВІ = ['Пошуки', 'Кандидати', 'Історія', 'Журнал', 'Інструкція'];

/** Google не дозволяє в назві аркуша деякі символи, і назва не може повторювати службову */
function безпечнаНазва(назва) {
  var чиста = String(назва || '').replace(/[\[\]\*\?\/\\:]/g, '-').trim().slice(0, 90);
  if (!чиста) чиста = 'Без назви';
  if (СЛУЖБОВІ.indexOf(чиста) !== -1) чиста += ' (вакансія)';
  return чиста;
}

/**
 * Повертає аркуш цієї вакансії, створюючи його за потреби.
 * Колонки беруться з «Кандидатів», тож структура завжди однакова —
 * включно з «Мій статус» і «Мої нотатки».
 */
function аркушВакансії(назваПошуку) {
  var ss = SpreadsheetApp.getActive();
  var назва = безпечнаНазва(назваПошуку);

  var наявний = ss.getSheetByName(назва);
  if (наявний) return наявний;

  var зразок = шапка(НАЗВИ.кандидати);
  var заголовки = зразок.sh
    .getRange(зразок.рядок, 1, 1, зразок.sh.getLastColumn())
    .getValues()[0];

  var sh = ss.insertSheet(назва);
  var зона = sh.getRange(1, 1, 1, заголовки.length);
  зона.setValues([заголовки]);
  зона.setFontWeight('bold').setBackground('#efefef');
  sh.setFrozenRows(1);

  // Щоб можна було сортувати за балом одним кліком
  try {
    sh.getRange(1, 1, sh.getMaxRows(), заголовки.length).createFilter();
  } catch (e) { /* фільтр уже є або аркуш не дозволяє — не критично */ }

  return sh;
}

function зробитиРядок(карта, значення) {
  var ширина = 0;
  for (var к in карта) ширина = Math.max(ширина, карта[к] + 1);

  var рядок = new Array(ширина).fill('');
  for (var назва in значення) {
    if (карта.hasOwnProperty(назва)) рядок[карта[назва]] = значення[назва];
  }
  return рядок;
}

function дайПошуки() {
  var ш = шапка(НАЗВИ.пошуки);
  var к = ш.колонки;
  var останній = ш.sh.getLastRow();
  if (останній <= ш.рядок) return [];

  var дані = ш.sh.getRange(ш.рядок + 1, 1, останній - ш.рядок, ш.sh.getLastColumn()).getValues();
  var список = [];

  for (var i = 0; i < дані.length; i++) {
    var р = дані[i];
    var назва = String(р[к['Назва пошуку']] || '').trim();
    var адреса = String(р[к['Посилання на пошук (з адресного рядка work.ua)']] || '').trim();
    var активний = String(р[к['Активний']] || '').trim().toLowerCase();

    if (!назва || !адреса) continue;
    if (['так', 'yes', 'y', '+', '1', 'true'].indexOf(активний) === -1) continue;

    список.push({
      назва: назва,
      адреса: адреса,
      ідеальний: String(р[к['Ідеальний кандидат — опиши своїми словами']] || '').trim(),
      стопСигнали: String(р[к['Стоп-сигнали — що дискваліфікує']] || '').trim(),
      мінімальнийБал: Number(р[к['Мінімальний бал для запису']]) || 0
    });
  }
  return список;
}

function оновитиСтатус(назваПошуку, текстСтатусу) {
  var ш = шапка(НАЗВИ.пошуки);
  var к = ш.колонки;
  var останній = ш.sh.getLastRow();
  if (останній <= ш.рядок) return;

  var стовпчикНазв = ш.sh
    .getRange(ш.рядок + 1, к['Назва пошуку'] + 1, останній - ш.рядок, 1)
    .getValues();

  for (var i = 0; i < стовпчикНазв.length; i++) {
    if (String(стовпчикНазв[i][0]).trim() !== назваПошуку) continue;

    var номерРядка = ш.рядок + 1 + i;
    if (к['Останній запуск'] !== undefined) {
      ш.sh.getRange(номерРядка, к['Останній запуск'] + 1).setValue(зараз());
    }
    if (к['Статус останнього запуску'] !== undefined) {
      ш.sh.getRange(номерРядка, к['Статус останнього запуску'] + 1).setValue(текстСтатусу);
    }
    return;
  }
}

function дайІсторію(назваПошуку) {
  var ш = шапка(НАЗВИ.історія);
  var к = ш.колонки;
  var останній = ш.sh.getLastRow();
  if (останній <= ш.рядок) return [];

  var дані = ш.sh.getRange(ш.рядок + 1, 1, останній - ш.рядок, ш.sh.getLastColumn()).getValues();
  var номери = [];

  for (var i = 0; i < дані.length; i++) {
    var посилання = String(дані[i][к['Посилання на резюме']] || '');
    var зПошуку = String(дані[i][к['З якого пошуку']] || '').trim();

    // Дублі рахуємо в межах однієї вакансії: та сама людина може
    // бути доречною для іншої позиції
    if (назваПошуку && зПошуку && зПошуку !== назваПошуку) continue;

    var m = посилання.match(/\/resumes\/(\d+)/);
    if (m) номери.push(m[1]);
  }
  return номери;
}

/**
 * Прибирає з «Історії» рядки однієї вакансії — щоб агент міг оцінити
 * тих самих людей заново (наприклад, після того як переписали опис
 * ідеального кандидата).
 *
 * Історія інших вакансій не чіпається. Аркуш «Кандидати» теж не чіпається:
 * там можуть бути твої нотатки, і вирішувати їхню долю має людина.
 *
 * Працює через перезапис, а не через видалення рядків по одному —
 * інакше на кількох тисячах рядків Apps Script не встигне за свої 6 хвилин.
 */
function очиститиІсторію(назваПошуку) {
  if (!назваПошуку) throw new Error('Не вказано, яку вакансію чистити');

  var ш = шапка(НАЗВИ.історія);
  var к = ш.колонки;
  var останній = ш.sh.getLastRow();
  if (останній <= ш.рядок) return 0;

  var висота = останній - ш.рядок;
  var ширина = ш.sh.getLastColumn();
  var дані = ш.sh.getRange(ш.рядок + 1, 1, висота, ширина).getValues();

  var лишити = дані.filter(function (р) {
    return String(р[к['З якого пошуку']]).trim() !== назваПошуку;
  });

  var видалено = дані.length - лишити.length;
  if (!видалено) return 0;

  ш.sh.getRange(ш.рядок + 1, 1, висота, ширина).clearContent();
  if (лишити.length) {
    ш.sh.getRange(ш.рядок + 1, 1, лишити.length, ширина).setValues(лишити);
  }

  return видалено;
}

/**
 * кандидати: масив обʼєктів
 *   { номер, посилання, бал, чому, посада, місто, вік, досвід, зарплата, записати }
 * «записати: false» означає — до «Кандидатів» не потрапляє (не пройшов
 * за балом), але в «Історію» все одно йде, щоб удруге його не оцінювати.
 */
function зберегти(назваПошуку, кандидати) {
  if (!кандидати.length) return { записано: 0, аркуш: '' };

  var дата = зараз();

  var шК = ОКРЕМИЙ_АРКУШ_НА_ВАКАНСІЮ
    ? шапкаЗа(аркушВакансії(назваПошуку), ЯКОРІ[НАЗВИ.кандидати], назваПошуку)
    : шапка(НАЗВИ.кандидати);

  var шІ = шапка(НАЗВИ.історія);

  var рядкиКандидатів = [];
  var рядкиІсторії = [];

  for (var i = 0; i < кандидати.length; i++) {
    var к = кандидати[i];

    if (к.записати) {
      рядкиКандидатів.push(зробитиРядок(шК.колонки, {
        'Дата знахідки':         дата,
        'Пошук':                 назваПошуку,
        'Бал':                   к.бал,
        'Чому такий бал':        к.чому,
        'Посада':                к.посада,
        'Місто':                 к.місто,
        'Вік':                   к.вік,
        'Досвід':                к.досвід,
        'Зарплатні очікування':  к.зарплата,
        'Посилання на резюме':   к.посилання
        // «Мій статус» і «Мої нотатки» лишаються порожніми — вони твої
      }));
    }

    рядкиІсторії.push(зробитиРядок(шІ.колонки, {
      'Посилання на резюме': к.посилання,
      'Дата першої появи':   дата,
      'З якого пошуку':      назваПошуку
    }));
  }

  if (рядкиКандидатів.length) {
    шК.sh.getRange(шК.sh.getLastRow() + 1, 1, рядкиКандидатів.length, рядкиКандидатів[0].length)
      .setValues(рядкиКандидатів);
  }
  if (рядкиІсторії.length) {
    шІ.sh.getRange(шІ.sh.getLastRow() + 1, 1, рядкиІсторії.length, рядкиІсторії[0].length)
      .setValues(рядкиІсторії);
  }

  return { записано: рядкиКандидатів.length, аркуш: шК.sh.getName() };
}

function записатиЖурнал(р) {
  var ш = шапка(НАЗВИ.журнал);
  var рядок = зробитиРядок(ш.колонки, {
    'Дата й час':               зараз(),
    'Пошук':                    р.пошук || '',
    'Сторінок пройдено':        р.сторінок || 0,

    // Відрізаємо шапку сайту й хвіст зі схожими резюме
    var початок = текст.indexOf('Резюме від');
    if (початок > 0) текст = текст.slice(початок);

    ['Схожі резюме', 'Ще резюме', 'Порівняти з іншими'].forEach(function (мітка) {
      var і = текст.indexOf(мітка);
      if (і > 500) текст = текст.slice(0, і);
    });

    return прибратиОсобисте(текст).slice(0, 4000);
  }

  /**
   * Страхувальна сітка: у Gemini не має потрапити нічого контактного,
   * навіть якщо work.ua колись почне це показувати.
   */
  function прибратиОсобисте(текст) {
    return String(текст)
      .replace(/[\w.+-]+@[\w-]+\.[\w.]+/g, '[пошта прихована]')
      .replace(/(?:\+?38)?[\s(-]*0\d{2}[\s)-]*\d{3}[\s-]*\d{2}[\s-]*\d{2}/g, '[телефон прихований]')
      .replace(/Прізвище, контакти та світлина[^.]*\./g, '');
  }

  async function оцінити(пошук, люди, повніРезюме) {
    var результат = await доGemini(підказка(пошук, люди, повніРезюме), СХЕМА);
    var заНомером = {};

    (результат || []).forEach(function (р) {
      if (р && р.id) заНомером[String(р.id).trim()] = р;
    });

    return люди.map(function (л) {
      var о = заНомером[л.номер];
      if (!о) return null;
      return {
        номер:      л.номер,
        посилання:  л.посилання,
        картка:     л.картка || '',
        бал:        Math.max(0, Math.min(100, Number(о.score) || 0)),
        чому:       о.why || '',
        посада:     о.position || '',
        місто:      о.city || '',
        вік:        о.age || '',
        досвід:     о.experience || '',
        зарплата:   о.salary || ''
      };
    });
  }

  var зупинити = false;

  /**
   * Обхід ведеться не «до сторінки N», а «поки не набереться N НОВИХ людей».
   *
   * Чому не по сторінках: work.ua сортує резюме за свіжістю, і люди додають
   * їх щогодини. За ніч список сповзає вниз на цілу сторінку, тож учорашня
   * закладка «сторінка 3» сьогодні вказує вже на інших людей — а найсвіжіші,
   * що з'явились уночі вгорі, агент не побачив би взагалі.
   *
   * Тому агент завжди починає з першої сторінки. Тих, хто вже в «Історії»,
   * він упізнає і пропускає — це не коштує ні запиту до Gemini, ні відкриття
   * резюме, тільки читання сторінки. Гортає доти, доки не набере потрібну
   * кількість справді нових.
   */
  async function запустити(пошук, скількиНових) {
    var н = налаштування();
    зупинити = false;

    var лічильники = {
      сторінок: 0,
      переглянуто: 0,   // усе, що трапилось на сторінках
      пропущено: 0,     // з них — уже в «Історії»
      нових: 0,         // з них — узяті в роботу
      оцінено: 0,       // з них — дійшли до оцінки
      записано: 0       // з них — пройшли за мінімальним балом
    };
    var помилки = [];

    журнал('Пошук: ' + пошук.назва);
    статус('Питаю таблицю, кого вже бачили…');

    var історія = (await доТаблиці('історія', { пошук: пошук.назва })).номери || [];
    var вжеБачили = {};
    історія.forEach(function (н) { вжеБачили[н] = true; });
    журнал('В «Історії» вже ' + історія.length + ' кандидатів — їх пропускаємо.');

    статус('Читаю першу сторінку…');
    var перша = await сторінка(адресаСторінки(пошук.адреса, 1));
    var всього = скількиСторінок(перша);

    var стеля = Number(н.максСторінокЗаЗапуск) || 0;
    var межа = стеля > 0 ? Math.min(стеля, всього) : всього;

    журнал('Усього сторінок: ' + всього + '. Шукаю ' + скількиНових + ' нових кандидатів.');

    var черга = [];
    var обірвано = false;

    async function оцінитиПорцію(порція) {
      try {
        лічильники.записано += await обробити(пошук, порція, н, помилки, лічильники);
        return true;
      } catch (e) {
        помилки.push(e.message);
        журнал('⛔️ ' + e.message);
        журнал('Ці ' + порція.length + ' кандидатів не втрачені: їх немає в «Історії», ' +
               'тож наступний запуск підбере їх заново.');
        обірвано = true;
        return false;
      }
    }

    for (var с = 1; с <= межа; с++) {
      if (зупинити) { журнал('Зупинено на сторінці ' + с + '.'); break; }

      var doc;
      try {
        doc = (с === 1) ? перша : await сторінка(адресаСторінки(пошук.адреса, с));
      } catch (e) {
        помилки.push('Сторінка ' + с + ': ' + e.message);
        журнал('⚠️ ' + e.message);
        break;
      }

      var картки = карткиЗіСторінки(doc);
      лічильники.сторінок++;
      лічильники.переглянуто += картки.length;

      if (картки.length === 0) {
        помилки.push('Сторінка ' + с + ': жодного резюме — можливо, змінилась верстка');
        журнал('⚠️ Сторінка ' + с + ': порожньо.');
      }

      картки.forEach(function (к) {
        if (лічильники.нових >= скількиНових) return;   // набрали скільки треба
        if (вжеБачили[к.номер]) { лічильники.пропущено++; return; }
        вжеБачили[к.номер] = true;
        черга.push(к);
        лічильники.нових++;
      });

      статус('Сторінка ' + с + ' · нових ' + лічильники.нових + ' з ' + скількиНових +
             ' · записано ' + лічильники.записано);

      // Оцінюємо по ходу, щоб результат зʼявлявся в таблиці, не чекаючи кінця
      while (черга.length >= н.пачкаКарток * 2) {
        if (!await оцінитиПорцію(черга.splice(0, н.пачкаКарток * 2))) break;
      }
      if (обірвано) break;

      if (лічильники.нових >= скількиНових) {
        журнал('Набрано ' + скількиНових + ' нових на сторінці ' + с + '. Далі не гортаю.');
        break;
      }

      if (с === межа && межа < всього) {
        журнал('⚠️ Дійшла до межі в ' + межа + ' сторінок за запуск, а нових набралось ' +
               лічильники.нових + '. Підніми межу в налаштуваннях або звузь фільтр на work.ua.');
      }

      if (с < межа) await пауза(н.паузаСторінки);
    }

    if (черга.length && !зупинити && !обірвано) {
      await оцінитиПорцію(черга.splice(0, черга.length));
    }

    // Розкладаємо підсумок так, щоб числа сходились і не треба було гадати,
    // куди поділась різниця між «побачив» і «записав»
    var недоглянуті = Math.max(0,
      лічильники.переглянуто - лічильники.пропущено - лічильники.нових);
    var відсіяні = Math.max(0, лічильники.оцінено - лічильники.записано);

    журнал('— — —');
    журнал('Трапилось на сторінках: ' + лічильники.переглянуто);
    if (лічильники.пропущено) {
      журнал('  уже в «Історії», пропущені безкоштовно: ' + лічильники.пропущено);
    }
    if (недоглянуті) {
      журнал('  не дивився — вже набрав потрібну кількість: ' + недоглянуті);
    }
    журнал('  оцінено через Gemini: ' + лічильники.оцінено);
    if (відсіяні) {
      журнал('    з них нижче мінімального балу ' + пошук.мінімальнийБал + ': ' + відсіяні);
    }
    журнал('  записано в таблицю: ' + лічильники.записано);

    var підсумок = 'Готово · оцінено ' + лічильники.оцінено +
                   ', записано ' + лічильники.записано +
                   (лічильники.пропущено ? ', пропущено знайомих ' + лічильники.пропущено : '');

    статус(підсумок);
    журнал(підсумок);

    await доТаблиці('журнал', {
      рядок: {
        пошук: пошук.назва,
        сторінок: лічильники.сторінок,
        переглянуто: лічильники.переглянуто,
        нових: лічильники.нових,
        записано: лічильники.записано,
        помилки: помилки.slice(0, 5).join(' | ')
      }
    });
    await доТаблиці('статус', { пошук: пошук.назва, статус: підсумок });
  }

  /** Два проходи над однією порцією кандидатів */
  async function обробити(пошук, порція, н, помилки, лічильники) {
    статус('Оцінюю ' + порція.length + ' кандидатів…');

    // ── Прохід 1: коротко, за картками ─────────────────────────────
    var чорнові = [];
    for (var i = 0; i < порція.length; i += н.пачкаКарток) {
      if (зупинити) break;
      var шматок = порція.slice(i, i + н.пачкаКарток);
      var оцінки = await оцінити(пошук, шматок, false);

      оцінки.forEach(function (о, j) {
        if (о) чорнові.push(о);
        else помилки.push('Gemini пропустив кандидата ' + шматок[j].номер);
      });
      журнал('Чорнова оцінка: ' + Math.min(i + н.пачкаКарток, порція.length) +
             ' з ' + порція.length);
    }

    // ── Прохід 2: повні резюме тих, хто пройшов поріг ──────────────
    var доУточнення = чорнові.filter(function (о) { return о.бал >= н.відкриватиРезюмеВід; });
    журнал('Варті повного резюме: ' + доУточнення.length + ' з ' + чорнові.length);

    var уточнені = {};
    for (var k = 0; k < доУточнення.length; k += н.пачкаРезюме) {
      if (зупинити) break;
      var група = доУточнення.slice(k, k + н.пачкаРезюме);
      var зТекстом = [];

      for (var g = 0; g < група.length; g++) {
        try {
          група[g].повне = await текстРезюме(група[g].посилання);
          зТекстом.push(група[g]);
        } catch (e) {
          // Резюме не відкрилось — людина лишається з чорновим балом,
          // вигадувати за неї нічого не будемо
          помилки.push('Резюме ' + група[g].номер + ': ' + e.message);
        }
        await пауза(н.паузаСторінки);
      }

      if (!зТекстом.length) continue;

      var точні = await оцінити(пошук, зТекстом, true);
      точні.forEach(function (о) { if (о) уточнені[о.номер] = о; });
      статус('Уточнено ' + Object.keys(уточнені).length + ' з ' + доУточнення.length);
    }

    // ── Складаємо в таблицю ────────────────────────────────────────
    var доЗапису = чорнові.map(function (о) {
      var фінальний = уточнені[о.номер] || о;
      return {
        номер:      фінальний.номер,
        посилання:  фінальний.посилання,
        бал:        фінальний.бал,
        чому:       фінальний.чому,
        посада:     фінальний.посада,
        місто:      фінальний.місто,
        вік:        фінальний.вік,
        досвід:     фінальний.досвід,
        зарплата:   фінальний.зарплата,
        записати:   фінальний.бал >= пошук.мінімальнийБал
      };
    });

    if (!доЗапису.length) return 0;

    лічильники.оцінено += доЗапису.length;

    var в = await доТаблиці('зберегти', { пошук: пошук.назва, кандидати: доЗапису });
    var відсіяно = доЗапису.length - в.записано;

    журнал('Записано ' + в.записано + (в.аркуш ? ' → аркуш «' + в.аркуш + '»' : '') +
           (відсіяно ? '; ' + відсіяно + ' не дотягнули до балу ' + пошук.мінімальнийБал : ''));
    return в.записано;
  }

Скільки це коштує

Нічого. Ані таблиця, ані Tampermonkey, ані ключ Gemini грошей не просять, картку вводити не треба.

Безкоштовний тариф дає 500 звернень на добу. Помічник складає кандидатів у пачки, тож одне звернення — приблизно семеро людей. Виходить близько 3 400 резюме на день, і на це піде години дві.

Комп'ютер при цьому має бути ввімкнений, Chrome запущений, а вкладка з work.ua відкрита — помічник живе в ній.

Про персональні дані

Контакти помічник не збирає. Прізвище, телефон, пошта й світлина доступні лише роботодавцю за підпискою — у таблицю йде тільки посилання на резюме. Контакти відкриваєш сама, вручну, і лише для тих, кому дзвонитимеш.

У Gemini йде тільки професійна частина — досвід, освіта, навички. Перед відправкою помічник додатково вирізає з тексту все, що схоже на телефон або пошту.

Але на безкоштовному тарифі Google має право використовувати надіслане для покращення своїх моделей — так написано в їхніх умовах. На платному тарифі ні. Якщо працюєш із чутливими вакансіями, це варто врахувати.
